\subsection*{Fórmules}
\begin{center}
\renewcommand{\arraystretch}{2.3}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Pressió} & $p = \dfrac{F}{A}$ \qquad $p_{abs} = p_{rel} + p_{atm}\approx p_{rel} + \SI{1}{\bar}$\\
\textbf{Força d'un cilindre} & $F = p\cdot A$ \qquad $A = \dfrac{\pi D^2}{4}$ \qquad corona: $A' = \dfrac{\pi (D^2 - d^2)}{4}$\\
\textbf{Boyle--Mariotte} ($T$ cte.) & $p_1 V_1 = p_2 V_2$\\
\textbf{Gay-Lussac} ($V$ cte.) & $\dfrac{p_1}{T_1} = \dfrac{p_2}{T_2}$\\
\textbf{Charles} ($p$ cte.) & $\dfrac{V_1}{T_1} = \dfrac{V_2}{T_2}$\\
\textbf{Llei general} & $\dfrac{p_1 V_1}{T_1} = \dfrac{p_2 V_2}{T_2}$ \qquad ($p$ absoluta, $T$ en kelvin)\\
\textbf{Pascal} & $\dfrac{F_1}{A_1} = \dfrac{F_2}{A_2}$ \qquad $A_1 s_1 = A_2 s_2$\\
\textbf{Treball} & $W = F\cdot s$\\
\bottomrule
\end{tabular}
\end{center}
